```cpp
#include<iostream>
#include<string>
#include<unordered_map>

using namespace std;
typedef long long ll;

const int N = 3e4 + 10, base = 13331, mod = 1e9 + 3;
int n, l, _, nn; ll ans = 0;
string s[N]; unordered_map<int, int> mp; 

int ctox(char c){
	if('a' <= c && c <= 'z') return c - 'a' + 1;
	else if('A' <= c && c <= 'Z') return c - 'A' + 27;
	else if('0' <= c && c <= '9') return c - '0' + 53;
	else if(c == '_') return 63;
	else if(c == '@') return 64;
	return 65;
}

int hs(string s){
	int res = 0;
	for(int i = 0; i < l; i++){
		res = ((ll)res * base + ctox(s[i])) % mod;
	}
	return res;
}

int main(){
	cin >> n >> l >> _;
	for(int i = 0; i < n; i++){
		cin >> s[i];
	}
	for(int j = 0; j < l; j++){
		for(int i = 0; i < n; i++){
			char c = s[i][j];
			s[i][j] = '#';
			mp[hs(s[i])]++;
			s[i][j] = c;
		}
		for(auto i: mp){
			ans += (ll)i.second * (i.second-1) / 2;
		}
		mp.clear();
	}
	cout << ans;
	return 0;
}
```

```cpp
#include<iostream>
#include<string>

using namespace std;
typedef long long ll;

const ll N = 2e5 + 10, mod = 1e9 + 7;
ll n, q, f[N][70], g[N][70];
ll m, s, k;
string ss;

int main(){
//	freopen("kingdom.in", "r", stdin);
//	freopen("kingdom.out", "w", stdout);
	cin >> n >> m >> q >> ss;
	for(ll i = 0; i < n; i++){
		f[i][0] = -1;
	}
	ll u = -1;
	for(ll i = (m%n)+n-1+((m%n==0)*n); i >= 0; i--){
		if(u != -1) while(u - i > m%n+((m%n==0)*n) || ss[u%n] == '0'){
			u--;
			if(u == i){
				u = -1; break;
			}
		}
		if(m > n && ss[i%n] == '1' && u == -1){
			u = i;
		}
//		cout << i << ' ' << u << '\n';
		if(u != -1 && u - i <= m%n+((m%n==0)*n) && i < n){
			if(u >= n) g[i][0] = 1, f[i][0] = u-n;
			else f[i][0] = u;
		}
		if(ss[i%n] == '1' && u == -1){
			u = i;
		}
	}
	ll bk = -1;
	for(ll i = 0; i < n; i++){
		if(ss[i] == '1') bk = i;
	}
	for(ll i = 0; i < n; i++){
		if(f[i][0] == -1){
			if(m > n){
				if(bk == -1){
					f[i][0] = i+1;
					if(f[i][0] == n) f[i][0] = 0, g[i][0] = 1;
				}
				else{
					f[i][0] = bk;
					g[i][0] = m/n-1;
				}
			}
			else{
				f[i][0] = i+1;
				if(f[i][0] == n) f[i][0] = 0, g[i][0] = 1;
			}
		}
		else g[i][0] = (m/n-(m%n==0) + g[i][0]) % mod;
	}
	for(ll j = 1; j < 70; j++){
		for(ll i = 0; i < n; i++){
			f[i][j] = f[f[i][j-1]][j-1];
			g[i][j] = (g[i][j-1] + g[f[i][j-1]][j-1]) % mod;
		}
	}
	while(q--){
		cin >> s >> k; s--;
		ll cnt = s / n % mod;
		s %= n;
		for(ll i = 60; i >= 0; i--){
			if(k & (1ll << i)){
				cnt = (g[s][i] + cnt) % mod;
				s = f[s][i];
			}
		}
		cout << ((ll)cnt*n%mod+s+1) % mod << '\n';
	}
	return 0;
}
```

```cpp
#include<iostream>
#include<string>
#include<vector>

using namespace std;

int s_to_i(char s){
	if (s >= 'a' && s <= 'z'){
		return s - 'a';
	}
}
const int N = 1e6 + 10, M = 1e5 + 10;

struct Tree{
	int data[N][26], id, cnt[N], end[N];
	vector<int> vec[N];
	
	void setup(){
		for(int i = 0; i < N; i++){
			for(int j = 0; j < 26; j++){
				data[i][j] = 0;
			}
			cnt[i] = 0;
			end[i] = -1;
		}
		id = 0;
	}
	
	void insert(string s, int idx){
		int cur = 0;
		for(int i = 0; i < s.size(); i++){
			int p = s_to_i(s[i]);
			if(data[cur][p] == 0) data[cur][p] = ++id;
			cur = data[cur][p];
			vec[cur].push_back(idx);
		}
		end[cur] = idx;
	}
	
	string query(string s){
		int cur = 0;
		for(int i = 0; i < s.size(); i++){
			int p = s_to_i(s[i]);
			if(data[cur][p] == 0){
				return "N";
			}
			cur = data[cur][p];
		}
		int num = vec[cur].size();
		while(true){
			bool flag = false;
			int tmp = -1;
			for(int i = 0; i < 26; i++){
				if(data[cur][i] != 0 && tmp == -1) tmp = i;
				else if(data[cur][i] != 0) flag = true;
			}
			cur = data[cur][tmp];
			if(flag || tmp == -1 || num != vec[cur].size()) break;
			else s += (tmp + 'a');
		}
		return s;
	}
	
	int find_str(string s){
		int cur = 0;
		for(int i = 0; i < s.size(); i++){
			int p = s_to_i(s[i]);
			if(data[cur][p] == 0){
				return -1;
			}
			cur = data[cur][p];
		}
		return end[cur];
	}
};

Tree tr;
int n, m;
string s, p, tmp;

int main(){
	tr.setup();
	cin >> n >> m;
	for(int i = 1; i <= n; i++){
		cin >> s;
		tr.insert(s, i);
	}
	cin >> s;
	for(int i = 0; i < s.size(); i++){
		if(s[i] >= 'a' && s[i] <= 'z'){
			p += s[i];
		}
		else if(s[i] == 'B'){
			if(p.size() > 0) p = p.substr(0, p.size() - 1); 
		}
		else if(s[i] == 'T'){
			tmp = tr.query(p);
			if(tmp != "N") p = tmp;
		}
		else if(s[i] == 'E'){
			cout << tr.find_str(p) << ' ';
			p = "";
		}
//		cout << p << endl;
	}
	return 0;
}
```
